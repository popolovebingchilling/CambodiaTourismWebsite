ព័ត៌មានវីសា និងរូបិយប័ណ្ណ

រៀបចំលិខិតសម្រាប់ចូលប្រទេស និងយល់ដឹងពីប្រព័ន្ធរូបិយប័ណ្ណក្នុងកម្ពុជា

ព័ត៌មានវីសា (Visa)

  • ទទួលបាន Visa on Arrival នៅព្រលានយន្តហោះ និងច្រកព្រំដែនសំខាន់ៗ
  • មានប្រភេទ e-Visa ដាក់ពាក្យតាមអ៊ីនធឺណិត មុនមកដល់
  • រយៈពេលស្នាក់នៅធម្មតា៖ 30 ថ្ងៃ
  • ត្រូវមានលិខិតឆ្លងដែនមានសុពលភាពយ៉ាងហោចណាស់ 6 ខែ

រូបិយប័ណ្ណ (Riel / USD)

  • រូបិយប័ណ្ណផ្លូវការ៖ រៀល (KHR)
  • ប៉ុន្តែ ដុល្លារអាមេរិក (USD) ត្រូវបានប្រើប្រាស់យ៉ាងទូលំទូលាយ
  • ការទូទាត់តូចៗ ឬប្រាក់ប្តូរ ស្ទើរតែប្រើរៀល
  • មាន ATM និងធនាគារច្រើន នៅទីក្រុងធំដូចជា ភ្នំពេញ និង សៀមរាប
កាលវិភាគ

ពេលល្អបំផុតសម្រាប់ដំណើរកម្សាន្ត

កម្ពុជាមានបីរដូវ ប៉ុន្តែរដូវត្រជាក់គឺជាពេលល្អបំផុតសម្រាប់ការធ្វើដំណើរទេសចរណ៍។

អាកាសធាតុអាក់អន់ មិនក្តៅខ្លាំង
មិនមានភ្លៀងរាំងងាយស្រួលធ្វើដំណើរ
សម្រាប់ទស្សនាប្រាសាទ និងធម្មជាតិ

កាលវិភាគរដូវកាល

ខែវិច្ឆិកា — ខែមីនា ពេលល្អបំផុត
អាកាសធាតុអាក់អន់ ត្រជាក់ស្រួល គ្មានភ្លៀងធ្លាក់ច្រើន — ពេលល្អបំផុតសម្រាប់ទេសចរណ៍
ខែមេសា — ខែមិថុនា
ក្តៅខ្លាំង សីតុណ្ហភាពអាចលើស ៤០°C ត្រូវជៀសវាងដំណើរយូរ
ខែកក្កដា — ខែតុលា
រដូវភ្លៀង ប៉ុន្តែធម្មជាតិស្រស់ស្អាត ទឹកត្រពាំងពេញ ស្មៅពណ៌ស្អាត
សុវត្ថិភាព និងទំនាក់ទំនង

វប្បធម៌ និងគន្លឹះសុវត្ថិភាព

គោរពសណ្តាប់ធម៌ និងការពារខ្លួនអ្នកក្នុងដំណើរ

វប្បធម៌ និងច្បាប់

  • ត្រូវគោរពវប្បធម៌ខ្មែរ និងសាសនា ព្រះពុទ្ធសាសនា
  • ពាក់សម្លៀកបំពាក់សមរម្យ ពេលចូលវត្ត ឬប្រាសាទ
  • មិនត្រូវប៉ះក្បាលអ្នកដទៃ (គេចាត់ទុកថាមិនគួរសម)
  • គោរពច្បាប់ និងចៀសវាងអំពើខុសច្បាប់
  • ការសំពះ (Sampeah) ជាវិធីសម្ដែងការគោរព

គន្លឹះសុវត្ថិភាព

  • ថែរក្សាសម្ភារៈផ្ទាល់ខ្លួន (ទូរស័ព្ទ កាបូប)
  • ជៀសវាងដើរតែម្នាក់ឯងពេលយប់ នៅកន្លែងស្ងាត់
  • ប្រើសេវាកម្មដឹកជញ្ជូនដែលទុកចិត្តបាន
  • មានលេខទំនាក់ទំនងបន្ទាន់ និងធានារ៉ាប់រងដំណើរ
  • ផឹកទឹកបរិសុទ្ធ និងជៀសវាងអាហារមិនស្អាត
សន្សំថវិកា

ដំណើរកម្សាន្តថវិកាតិច

អាចធ្វើដំណើរកម្ពុជាបានដោយថវិកាតិច ប្រសិនបើអ្នកដឹងពីគន្លឹះទាំងនេះ។

គន្លឹះថវិកាតិច

ជ្រើសរើស Guesthouse ឬ Hostel តម្លៃសមរម្យ
ប្រើ Tuk Tuk ឬ Bus ជំនួស Taxi
ញ៉ាំអាហារតាមផ្សារ ឬ Street Food តម្លៃទាប
កក់សំបុត្រដំណើរជាមុន ដើម្បីបានតម្លៃល្អ
ទៅកន្លែងទេសចរណ៍ ឥតគិតថ្លៃ ដូចជា សួនច្បារ ឬឆ្នេរ
អ្នកជំនាញក្នុងតំបន់

មគ្គុទ្ទេសក៍តាមបណ្តាខេត្ត

ជួបជាមួយអ្នកជំនាញក្នុងតំបន់ដែលត្រៀមខ្លួនជាស្រេច ដើម្បីនាំអ្នកទៅកាន់កន្លែងដែលល្អបំផុត និងរឿងរ៉ាវដែលអ្នកមិនធ្លាប់ដឹង។

សៀមរាប
អ្នកជំនាញប្រវត្តិសាស្ត្រ

ស្វែងយល់ពីអាថ៌កំបាំងនៃប្រាសាទបុរាណ ជាមួយមគ្គុទ្ទេសក៍ដែលមានជំនាញឯកទេសខាងបុរាណវិទ្យា។

កំពត
អ្នកជំនាញធម្មជាតិ

ដើរកម្សាន្តតាមចម្ការម្រេច និងភ្នំបូកគោ ជាមួយអ្នកជំនាញដែលស្គាល់គ្រប់ផ្លូវក្នុងខេត្តកំពត។

ព្រះសីហនុ
អ្នកជំនាញសមុទ្រ

ស្វែងរកកោះស្ងប់ស្ងាត់ និងតំបន់មុជទឹកមើលផ្កាថ្ម ជាមួយអ្នកជំនាញក្នុងតំបន់ឆ្នេរ។

មណ្ឌលគិរី
អ្នកជំនាញអេកូឡូស៊ី

បទពិសោធន៍រស់នៅជាមួយដំរី និងជនជាតិដើមភាគតិច ជាមួយមគ្គុទ្ទេសក៍ការពារបរិស្ថាន។

ភ្នំពេញ
អ្នកជំនាញវប្បធម៌

ស្វែងយល់ពីការវិវត្តនៃរាជធានីភ្នំពេញ ពីសម័យកាលប្រវត្តិសាស្ត្ររហូតដល់សម័យទំនើប។

បាត់ដំបង
អ្នកជំនាញសិល្បៈ

ស្វែងរកវិចិត្រសាលសិល្បៈ និងម្ហូបអាហារប្លែកៗ ជាមួយអ្នកជំនាញក្នុងក្រុងបាត់ដំបង។

កែប
អ្នកជំនាញសម្រាកកាយ

ទស្សនាផ្សារក្តាម និងឆ្នេរសមុទ្រស្ងប់ស្ងាត់ ជាមួយមគ្គុទ្ទេសក៍ដែលយល់ពីតម្រូវការសម្រាកកាយ។

កោះកុង
អ្នកជំនាញផ្សងព្រេង

រុករកព្រៃកោងកាង និងភ្នំក្រវាញ ជាមួយមគ្គុទ្ទេសក៍ដែលចូលចិត្តការផ្សងព្រេងពិតៗ។

រៀបចំដំណើររបស់អ្នកថ្ងៃនេះ

កម្ពុជារង់ចាំអ្នកជាមួយនឹងប្រាសាទបុរាណ ឆ្នេរសមុទ្រ និងអាហារស្រស់ថ្មី។

ចាប់ផ្តើមដំណើរ
ទេសចរណ៍កម្ពុជា

ស្វែងយល់ពីភាពអស្ចារ្យនៃព្រះរាជាណាចក្រអច្ឆរិយៈ។ យើងខ្ញុំផ្តល់ជូននូវបទពិសោធន៍ទេសចរណ៍ដ៏ល្អបំផុត ដើម្បីបំពេញក្តីសុបិនរបស់អ្នកនៅក្នុងតំបន់អាសុីអាគ្នេយ៍។
