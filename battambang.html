តថភាពមួយចំនួនទាក់ទងនឹងខេត្តបាត់ដំបង

ខេត្តបាត់ដំបង (Battambang Province) គឺជាខេត្តមួយស្ថិតនៅភាគខាងលិច-ខាងជើងនៃប្រទេសកម្ពុជា។ វាជាខេត្តសំខាន់ មួយទាំងផ្នែកសេដ្ឋកិច្ច វប្បធម៌ និងទេសចរណ៍។ ជាទូទៅ បាត់ដំបងគឺជាខេត្តស្ងប់ស្ងាត់ ប៉ុន្តែសម្បូរទៅដោយប្រវត្តិសាស្ត្រ វប្បធម៌ និងធម្មជាតិ ដែលគួរឲ្យទៅទស្សនា។

ប្រាសាទបុរាណខ្មែរ

ប្រាសាទឯកភ្នំ

ប្រាសាទអេកភ្នំ (Ek Phnom) គឺជាប្រាសាទបុរាណមួយដ៏ស្រស់ស្អាតនៅខេត្តបាត់ដំបង ដែលមានទាំងប្រវត្តិសាស្ត្រ និងទេសភាពគួរឱ្យចាប់អារម្មណ៍។

ទីតាំង៖ ប្រាសាទអេកភ្នំ ស្ថិតនៅក្នុងខេត្ត បាត់ដំបង ចម្ងាយ៖ ប្រហែល 13 គីឡូម៉ែត្រ ពីក្រុងបាត់ដំបង

ព័ត៍មានលម្អិត

ប្រាសាទឯកភ្នំ

លក្ខណៈសំខាន់ ប្រាសាទមានរចនាបថថ្មបុរាណ (laterite និង sandstone) នៅសល់តែផ្នែកខ្លះប៉ុណ្ណោះ ប៉ុន្តែមានអារម្មណ៍បុរាណខ្លាំង មានប្រាសាទតូចៗជុំវិញប្រាសាទចម្បងទេសភាពជុំវិញ វាលស្រែ និងភូមិជនបទខ្មែរ ខ្យល់បរិសុទ្ធ ស្ងប់ស្ងាត់ អាចជិះម៉ូតូ ឬ tuk-tuk ពីក្រុងបាត់ដំបង ផ្លូវងាយស្រួល និងមានស្លាកបង្ហាញ

ភ្នំសំពៅ

ភ្នំសំពៅ គឺជាទីតាំងទេសចរណ៍ធម្មជាតិ និងប្រវត្តិសាស្ត្រសំខាន់មួយនៅក្នុងប្រទេសកម្ពុជា ដែលមានទីតាំងនៅជិតទីក្រុង

ភ្នំសំពៅមានកម្ពស់ប្រហែល 100 ម៉ែត្រ ស្ថិតនៅចម្ងាយប្រហែល 12 គីឡូម៉ែត្រពីក្រុងបាត់ដំបង

ព័ត៍មានលម្អិត

ប្រាសាទសំពៅ

1. វត្តលើភ្នំ មានវត្តព្រះពុទ្ធសាសនា ដែលអ្នកអាចឡើងទៅសូមពរ អាចមើលទេសភាពទាំងមូលនៃខេត្តបាត់ដំបង 2. ល្អាងប្រវត្តិសាស្ត្រ (Killing Caves) ភ្នំសំពៅពាក់ព័ន្ធនឹងប្រវត្តិសាស្ត្រខ្មែរក្រហម នៅទីនេះមានល្អាងដែលត្រូវបានប្រើក្នុងអំឡុង របបខ្មែរក្រហម 3. ល្អាងសត្វប្រចៀវ (Bat Cave) ជាទីកន្លែងល្បីសម្រាប់មើលសត្វប្រចៀវរាប់ម៉ឺនក្បាលហោះចេញពេលល្ងាច ជាបទពិសោធន៍ធម្មជាតិដ៏អស្ចារ្យ

ប្រាសាទបាណន់

ប្រាសាទបាណន់ គឺជាប្រាសាទបុរាណដ៏ល្បីមួយនៅខេត្ត បាត់ដំបង ដែលមានស្ថាបត្យកម្មស្រស់ស្អាត និងទេសភាពល្អឥតខ្ចោះ។

មានប្រាសាទចម្បង 5 កំពូល (tower) សង់លើភ្នំតូចមួយ ត្រូវឡើងជណ្ដើរប្រហែល 350 ជំហាន ដើម្បីទៅដល់កំពូល រចនាប័ទ្មស្រដៀងនឹង

ព័ត៍មានលម្អិត

ប្រាសាទបាណន់

អ្វីដែលគួរទស្សនា 1. ទេសភាពពីលើភ្នំ អាចមើលឃើញវាលស្រែ និងភូមិជុំវិញយ៉ាងស្រស់ស្អាត 2. សំណង់បុរាណ ប្រាសាទមានការកសាងពីថ្ម laterite និង sandstone មានរចនាបថសិល្បៈខ្មែរបុរាណ 3. ទេសចរណ៍ធម្មជាតិជុំវិញ តំបន់មានដើមឈើ និងខ្យល់ត្រជាក់ល្អ ពេលវេលាល្អទៅ ព្រឹកឬល្ងាច (មិនក្តៅពេក) ពេលថ្ងៃលិច គឺស្អាតខ្លាំងសម្រាប់ថតរូប

អំពីផ្សារក្នុងទីរួមខេត្ត

ផ្សារ បវែលបាត់ដំបង

ផ្សារ បវែល គឺជាទីផ្សារមូលដ្ឋានមួយដែលស្ថិតនៅក្នុងស្រុកបវែល នៃខេត្ត Battambang ក្នុងប្រទេស Cambodia។ ទីផ្សារនេះមានតួនាទីសំខាន់ក្នុងជីវភាពប្រចាំថ្ងៃរបស់ប្រជាជនក្នុងតំបន់ ដោយជាកន្លែងដែលប្រជាជនមកជួញដូរ និងផ្លាស់ប្តូរទំនិញគ្រប់ប្រភេទចាប់ពីអាហាររហូតដល់សម្ភារៈប្រើប្រាស់ផ្សេងៗ។

ផ្សារ​ បវែល

អត្ថបទអំពីផ្សារ បវែល ខេត្តបាត់ដំបង ផ្សារ បវែល គឺជាទីផ្សារមូលដ្ឋានមួយដែលស្ថិតនៅក្នុងស្រុកបវែល នៃខេត្ត Battambang ក្នុងប្រទេស Cambodia។ ទីផ្សារនេះមានតួនាទីសំខាន់ក្នុងជីវភាពប្រចាំថ្ងៃរបស់ប្រជាជនក្នុងតំបន់ ដោយជាកន្លែងដែលប្រជាជនមកជួញដូរ និងផ្លាស់ប្តូរទំនិញគ្រប់ប្រភេទចាប់ពីអាហាររហូតដល់សម្ភារៈប្រើប្រាស់ផ្សេងៗ។ នៅពេលព្រឹកដំបូងៗ ផ្សារ បវែលចាប់ផ្តើមមានភាពរវល់យ៉ាងខ្លាំង ពេលដែលអ្នកលក់ និងអ្នកទិញមកប្រមូលផ្តុំគ្នា។ សកម្មភាពលក់ដូរនៅទីនេះមានភាពរស់រវើក ដោយមានសម្លេងនិយាយសន្ទនា ការចរចាតម្លៃ និងការហៅអតិថិជន ដែលបង្ហាញពីវប្បធម៌ទីផ្សារខ្មែរដែលសាមញ្ញ និងជិតស្និទ្ធ។ ទំនិញដែលលក់មានច្រើនប្រភេទដូចជា បន្លែ ផ្លែឈើស្រស់ៗ ដែលបានដាំដុះពីស្រុកជិតខាង សាច់ ត្រី និងអាហារស្រាលផ្សេងៗ ដែលបង្កើតភាពងាយស្រួលសម្រាប់ប្រជាជនក្នុងការរកទិញអាហារប្រើប្រាស់ប្រចាំថ្ងៃ។ មិនត្រឹមតែប៉ុណ្ណោះ ផ្សារ បវែលក៏មានលក់សម្លៀកបំពាក់ ស្បែកជើង និងសម្ភារៈប្រើប្រាស់ក្នុងផ្ទះផងដែរ។ តម្លៃទំនិញភាគច្រើនមានភាពសមរម្យ និងអាចចរចាបាន ដែលធ្វើឱ្យទីផ្សារនេះក្លាយជាជម្រើសសំខាន់សម្រាប់ប្រជាជនក្នុងតំបន់។ លក្ខណៈពិសេសមួយទៀតគឺភាពសាមញ្ញ និងភាពរួសរាយរបស់អ្នកលក់ ដែលធ្វើឱ្យអតិថិជនមានអារម្មណ៍ស្និទ្ធស្នាលដូចជាគ្រួសារ។ បើប្រៀបធៀបនឹងទីផ្សារធំៗដូចជា Phsar Nath ផ្សារ បវែលអាចមិនមានទំហំធំ ឬមានការរៀបចំទំនើបដូចនោះទេ ប៉ុន្តែវាមានតម្លៃខ្ពស់ផ្នែកសង្គម និងវប្បធម៌។ វាជាកន្លែងដែលបង្ហាញពីរបៀបរស់នៅពិតប្រាកដរបស់ប្រជាជនជនបទ និងជាកន្លែងដែលជួយបន្តទំនាក់ទំនងរវាងអ្នកនៅក្នុងសហគមន៍។ វាបង្ហាញពីភាពសាមញ្ញ ភាពរួសរាយ និងការរួមរស់រវាងមនុស្សក្នុងសង្គមខ្មែរ ដែលនៅតែរក្សាបានដល់សព្វថ្ងៃ។
